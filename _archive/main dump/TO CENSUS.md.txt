FROM
(SELECT  CASE_NO,  SOC_SEC_NO,
         EE_LAST_NM,
         EE_FST_MID_NM,
         MIDDLE_NM,
         SUFFIX_NM,
         CASE 
              WHEN substr (EE_BRTH_DT, 5, 2) || '/' || substr (EE_BRTH_DT, 7, 2) || '/' || substr (EE_BRTH_DT, 1, 4) = '00/00/0000' THEN '' 
              ELSE substr (EE_BRTH_DT, 5, 2) || '/' || substr (EE_BRTH_DT, 7, 2) || '/' || substr (EE_BRTH_DT, 1, 4) 
         END AS DATE_OF_BIRTH,
         CASE 
              WHEN substr (EE_HIRE_DT, 5, 2) || '/' || substr (EE_HIRE_DT, 7, 2) || '/' || substr (EE_HIRE_DT, 1, 4) = '00/00/0000' THEN '' 
              ELSE substr (EE_HIRE_DT, 5, 2) || '/' || substr (EE_HIRE_DT, 7, 2) || '/' || substr (EE_HIRE_DT, 1, 4) 
         END AS DATE_OF_HIRE,
         CASE 
              WHEN substr (EE_TERM_DT, 5, 2) || '/' || substr (EE_TERM_DT, 7, 2) || '/' || substr (EE_TERM_DT, 1, 4) = '00/00/0000' THEN '' 
              ELSE substr (EE_TERM_DT, 5, 2) || '/' || substr (EE_TERM_DT, 7, 2) || '/' || substr (EE_TERM_DT, 1, 4) 
         END AS DATE_OF_TERM,
         EE_REHIRE_DT,
         EE_PAYROLL_FREQ_CD,
         EE_SEX_CD, EE_CYC_DT, MOD_TS ,USER_I
FROM     CORP.EMPLOYEE
WHERE    CASE_NO = 'QK63283   00073'
--and soc_sec_no in ('591-66-2232','508-33-1880','520-23-9739','263-85-1822','599-46-7147')
) AS A

LEFT JOIN 

(SELECT CASE_NO, SOC_SEC_NO, STREET_NM, ADDR_LINE_2_T, CITY_NM, ST_CD, ZIP_CD, COUNTRY_NM, COUNTRY_C, REGION_ST_CD, PHONE_NO, EVE_PHONE_NO, MOBILE_PHONE_NO
FROM CORP.EE_BEN_ADDRESS
WHERE CASE_NO = 'QK63283   00073'
--and soc_sec_no in ('591-66-2232','508-33-1880','520-23-9739','263-85-1822','599-46-7147')
AND PAYEE_TYP_CD = '1') AS B

ON A.CASE_NO = B.CASE_NO AND  A.SOC_SEC_NO = B.SOC_SEC_NO

ORDER BY SOC_SEC_NO

FOR FETCH ONLY WITH UR
