DECLARE GLOBAL TEMPORARY TABLE SESSION.terminating_sub_ids
(
case_NO VARCHAR(20) NOT NULL,
eff_dt VARCHAR(10),
fd_Desc_cd VARCHAR(6),
fund_name VARCHAR(80),
SRC_TYP_C SMALLINT,
Conversion_amt DECIMAL(12,2)

	)WITH REPLACE ON COMMIT PRESERVE ROWS NOT LOGGED;


insert into session.terminating_sub_ids

with rundate as
(
SELECT CAST(MAX(To_DATE(HLP_VALUE,'YYYYMMDD')) AS DATE) as proc_date
FROM corp.HELPER2
WHERE HLP_CODE = 'BUSDAYS'
  AND CAST(To_DATE(HLP_VALUE,'YYYYMMDD') AS DATE) < current date
    AND HLP_TEXT LIKE '%OPEN%')


select CD.case_no, CAST((SELECT PROC_DATE FROM RUNDATE)AS VARCHAR),COALESCE(FD_DESCR_CD,'ACTIVE') AS FD_dESC_CD,COALESCE(FD_FUND_NAME,'NO BALANCE'), COALESCE(SRC_TYP_C,NULL) AS SRC_TYP_C,COALESCE(AMOUNT,0) AS CONVERSION_AMT

FROM CORP.CASE_DATA CD
 
LEFT JOIN (SELECT case_no, A.eff_d,FD_DESCR_CD, STRIP(STRIP(c.REPORT_1_FD_NM)||' '||STRIP(c.REPORT_2_FD_NM)) FD_FUND_NAME,SRC_TYP_C,sum(TOTAL_A) AS AMOUNT
from corp.ee_part_bal_header a, corp.ee_part_balance b
JOIN PENSION.PLAN_FUND c ON b.FD_PROV_I = c.FD_PROV_I
JOIN PENSION.PLAN_SRC_dETAIL D ON D.SRC_I=B.SRC_I
where  case_no like '$CASE%'
and substr(case_no,11,5)  in ($PLANS_TO_MOVE)
--and case_stat_cd='7'
and a.transact_i=b.transact_i
and a.enrl_prov_grp_i=b.enrl_prov_grp_i
and A.eff_d=(SELECT PROC_DATE FROM RUNDATE)

group by case_no, A.EFF_D,FD_DESCR_CD, STRIP(STRIP(c.REPORT_1_FD_NM)||' '||STRIP(c.REPORT_2_FD_NM)),SRC_TYP_C
) BAL
ON CD.CASE_NO=BAL.CASE_NO
WHERE CD.CASE_NO LIKE '$CASE %'
AND substr(CD.case_no,11,5)  in ($PLANS_TO_MOVE)

;



-- FUND BALANCE
select ' ' as new_case_no, '' as new_sub_id,SUM(CONVERSION_AMT) AS ESTIMATED_CONVERSION_AMT, 
EFF_DT, $CASE, FD_DESC_CD
--SELECT *
FROM SESSION.terminating_sub_ids
GROUP BY EFF_DT, FD_DESC_CD,FUND_NAME
having SUM(CONVERSION_AMT)>0

;

--- SOURCE BALANCE
select ' ' as new_case_no, '' as new_sub_id,SUM(CONVERSION_AMT) AS ESTIMATED_CONVERSION_AMT, 
  EFF_DT, $CASE, SRC_TYP_C
--SELECT *
FROM SESSION.terminating_sub_ids
GROUP BY EFF_DT,  SRC_TYP_C
having SUM(CONVERSION_AMT)>0

;

-- Individual sub_ID balances
select ' ' as new_case_no, '' as new_sub_id,SUM(CONVERSION_AMT) AS ESTIMATED_CONVERSION_AMT, 
  EFF_DT, $CASE, case_no
--SELECT *
FROM SESSION.terminating_sub_ids
GROUP BY EFF_DT,  case_no

;

--RMD

WITH	PLAN_LIST 
AS (
	SELECT 
	ACCOUNT_NO
	FROM PENSION.PLAN_PROV_GRP 
	WHERE RELATED_GRP_TYP_C = 361
	AND ACCOUNT_NO LIKE '$CASE%'
and account_no in (select case_no from session.terminating_sub_ids)
	AND ENRL_STAT_C = 7

	)

	,PYE_BALANCE
AS (
	SELECT *
	FROM (
		SELECT A.CASE_NO
			,A.SOC_SEC_NO
			,coalesce(B.PYE_ACCOUNT_BALANCE, LST_CAL_YR_END_BAL, 0) AS PYE_ACCOUNT_BALANCE
		FROM CORP.EMPLOYEE A
		JOIN PLAN_LIST PL ON A.CASE_NO = PL.ACCOUNT_NO
		LEFT OUTER JOIN (
			SELECT EFF_D
				,A.CASE_NO
				,B.SOC_SEC_NO
				,SUM(TOTAL_A) AS PYE_ACCOUNT_BALANCE
			FROM CORP.EE_PART_BAL_HEADER A
				,CORP.EE_PART_BALANCE B
				,PLAN_LIST C
			WHERE A.TRANSACT_I = B.TRANSACT_I
				AND A.CASE_NO = C.ACCOUNT_NO
				AND A.ENRL_PROV_GRP_I = B.ENRL_PROV_GRP_I
				AND A.EFF_D = DATE(TRIM(CHAR(YEAR(CURRENT DATE - 1 YEAR))) || '-12-31')
				AND SRC_CD = 1
			GROUP BY EFF_D
				,A.CASE_NO
				,B.SOC_SEC_NO
			) B ON B.CASE_NO = A.CASE_NO
			AND B.SOC_SEC_NO = A.SOC_SEC_NO
		LEFT OUTER JOIN (
			SELECT A.CASE_NO
				,A.SOC_SEC_NO
				,COALESCE(LST_CAL_YR_END_BAL, 0) LST_CAL_YR_END_BAL
			FROM CORP.MIN_DIST_DATA A
				,PLAN_LIST B
			WHERE EFF_YR = TRIM(CHAR(YEAR(CURRENT DATE - 1 YEAR)))
				AND A.CASE_NO = B.ACCOUNT_NO
			) C ON C.CASE_NO = A.CASE_NO
			AND C.SOC_SEC_NO = A.SOC_SEC_NO
		)
	WHERE PYE_ACCOUNT_BALANCE > 0
	)

	,CURRENT_BALANCE
AS (
	SELECT A.CASE_NO, B.SOC_SEC_NO, SUM(TOTAL_A) AS BALANCE, EFF_D
	FROM CORP.EE_PART_BAL_HEADER A, CORP.EE_PART_BALANCE B, PLAN_LIST C
	WHERE A.CASE_NO = C.ACCOUNT_NO
	AND A.TRANSACT_I = B.TRANSACT_I
	AND A.ENRL_PROV_GRP_I = B.ENRL_PROV_GRP_I
	AND A.EFF_D = (SELECT MAX(EFF_D) FROM CORP.EE_PART_BAL_HEADER)
	GROUP BY A.CASE_NO, B.SOC_SEC_NO, EFF_D
	)

SELECT 
'' as new_case_no, '' as new_sub_id,
substr(A.SOC_SEC_NO,1,3)||substr(A.SOC_SEC_NO,5,2)||substr(A.SOC_SEC_NO,8,4)  as SSN, 
EE_FST_MID_NM,
EE_LAST_NM,
DECIMAL(ROUND((DAYS(CURRENT DATE) - DAYS (DATE(SUBSTR(EE_BRTH_DT,1,4)||'-'||SUBSTR(EE_BRTH_DT,5,2)||'-'||SUBSTR(EE_BRTH_DT,7,2))))/365.24, 2),5,2) AGE,
BALANCE CURRENT_BALANCE,
COALESCE(PYE_ACCOUNT_BALANCE,0) PYE_BALANCE,substr(A.case_NO,1,instr(A.case_no,' ')-1) as old_cont,SUBSTR(A.CASE_NO,11,5) as old_sub_id
FROM CORP.EMPLOYEE A 
JOIN PLAN_LIST B ON A.CASE_NO = B.ACCOUNT_NO
LEFT JOIN PYE_BALANCE C ON A.CASE_NO = C.CASE_NO
	AND A.SOC_SEC_NO = C.SOC_SEC_NO
JOIN CURRENT_BALANCE D ON A.CASE_NO = D.CASE_NO
	AND A.SOC_SEC_NO = D.SOC_SEC_NO


;

--YTD CONTRIBS


with calendar_year as

(SELECT 
substr(b.case_NO,1,instr(b.case_no,' ')-1) as cont,SUBSTR(B.CASE_NO,11,5) CASE_NO,
SUBSTR(B.SOC_SEC_NO,1,11) AS SOC_sEC_NO,src_typ_c,c.src_i,
DOC_NM,CONTRIB_YR_D,
SUM(CONTRIB_A) AS CONTRIB_A
FROM PENSION.PLAN_PROV_GRP A
JOIN CORP.EMPLOYEE B ON A.ACCOUNT_NO = B.CASE_NO
JOIN PENSION.PART_CONTRIB C ON B.CASE_NO = C.ACCOUNT_NO
	AND B.SOC_SEC_NO = C.SOC_SEC_NO
	AND CONTRIB_YR_D = year(current date)
	AND CONTRIB_TYP_C = '1'  
JOIN PENSION.plan_src_detail D ON C.SRC_I = D.SRC_I
WHERE A.ACCOUNT_NO in (select case_no from session.terminating_sub_ids)
AND RELATED_GRP_TYP_C = 361
AND ENRL_STAT_C = 7

GROUP BY 
substr(b.case_NO,1,instr(b.case_no,' ')-1),SUBSTR(B.CASE_NO,11,5),SUBSTR(B.SOC_SEC_NO,1,11),src_typ_c,c.src_i,
DOC_NM,CONTRIB_YR_D

)

SELECT 
'' as New_case_no, '' as new_sub_id,
substr(b.SOC_SEC_NO,1,3)||substr(b.SOC_SEC_NO,5,2)||substr(b.SOC_SEC_NO,8,4) as SSN,c.CONTRIB_YR_D as plan_year, e.CONTRIB_YR_D as cal_year,'' as new_source_code,
SUM(c.CONTRIB_A) as plan_year, e.contrib_a as calendar_year,substr(b.case_NO,1,instr(b.case_no,' ')-1) as old_cont,SUBSTR(B.CASE_NO,11,5) old_CASE_NO,d.src_typ_c as old_source_code ,
d.DOC_Nm as old_source_name
FROM PENSION.PLAN_PROV_GRP A
JOIN CORP.EMPLOYEE B ON A.ACCOUNT_NO = B.CASE_NO
JOIN PENSION.PART_CONTRIB C ON B.CASE_NO = C.ACCOUNT_NO
	AND B.SOC_SEC_NO = C.SOC_SEC_NO
	AND c.CONTRIB_YR_D = year(current date)
	AND c.CONTRIB_TYP_C = '2' 
JOIN PENSION.plan_src_Detail D ON C.SRC_I = D.SRC_I
left join calendar_year e on   e.src_i=c.src_i and SUBSTR(e.soc_Sec_no,1,11)=SUBSTR(c.soc_Sec_no,1,11)
WHERE A.ACCOUNT_NO in (select case_no from session.terminating_sub_ids)
AND RELATED_GRP_TYP_C = 361
AND ENRL_STAT_C = 7
GROUP BY
substr(b.case_NO,1,instr(b.case_no,' ')-1) ,SUBSTR(B.CASE_NO,11,5) ,
substr(b.SOC_SEC_NO,1,3)||substr(b.SOC_SEC_NO,5,2)||substr(b.SOC_SEC_NO,8,4) ,d.src_typ_c,
d.DOC_NM,c.CONTRIB_YR_D,e.contrib_a, e.CONTRIB_YR_D




;

-- LOAN INFORMATION 1

where case_no in (select case_no from session.terminating_sub_ids)
and (LN_REPAY_STAT_CD='1' )
group by case_no, loan_no, soc_sec_no)

WITH YEAR_BACK
AS (
	SELECT DATE (SUBSTR(HLP_VALUE, 1, 4) || '-' || SUBSTR(HLP_VALUE, 5, 2) || '-' || SUBSTR(HLP_VALUE, 7, 2)) - 1 MONTH AS BEGIN_DATE
		,DATE (SUBSTR(HLP_VALUE, 1, 4) || '-' || SUBSTR(HLP_VALUE, 5, 2) || '-' || SUBSTR(HLP_VALUE, 7, 2)) AS END_DATE
	FROM CORP.HELPER2
	WHERE HLP_CODE = 'BUSDAYS' AND DATE (SUBSTR(HLP_VALUE, 1, 4) || '-' || SUBSTR(HLP_VALUE, 5, 2) || '-' || SUBSTR(HLP_VALUE, 7, 2)) BETWEEN CURRENT DATE - 12 MONTH AND CURRENT DATE AND SUBSTR(HLP_VALUE, 7, 2) = DAY(CURRENT DATE)
	)
	,PLAN_LIST
AS (
	SELECT ACCOUNT_NO
		,ENRL_PROV_GRP_I
		,ENRL_STAT_C
	FROM PENSION.PLAN_PROV_GRP
	WHERE ACCOUNT_NO in  (select case_no from session.terminating_sub_ids)
		AND RELATED_GRP_TYP_C = 361
		AND ENRL_STAT_C = 7
		
	)
	,LOAN_PAYMENTS
AS (
	SELECT BEGIN_DATE
		,END_DATE
		,B.*
	FROM YEAR_BACK AS A
	LEFT JOIN (
		SELECT A.CASE_NO
			,A.SOC_SEC_NO
			,B.LOAN_NO
			,B.EFF_DT
			,B.PRNC_BAL_DUE_AMT
		FROM (
			SELECT *
			FROM CORP.LOAN_DETAIL A
			JOIN PLAN_LIST B ON A.CASE_NO = B.ACCOUNT_NO
			WHERE 1=1
			AND LN_TR_STAT_CD NOT IN ('2', '3', '4', '5', '6')
			) AS A
		INNER JOIN (
			SELECT CASE_NO
				,SOC_SEC_NO
				,LOAN_NO
				,EFF_DT
				,MIN(PRNC_BAL_DUE_AMT) AS PRNC_BAL_DUE_AMT
			FROM (
				SELECT CASE_NO
					,SOC_SEC_NO
					,LOAN_NO
					,PRNC_BAL_DUE_AMT
					,(CAST(SUBSTR(EFF_DT, 1, 4) || '-' || SUBSTR(EFF_DT, 5, 2) || '-' || SUBSTR(EFF_DT, 7, 2) AS DATE)) AS EFF_DT
				FROM CORP.LOAN_REPAYMENT A
				JOIN PLAN_LIST B ON A.CASE_NO = B.ACCOUNT_NO
				
				UNION
				
				SELECT CASE_NO
					,SOC_SEC_NO
					,LOAN_NO
					,(INIT_LN_AMT) AS PRNC_BAL_DUE_AMT
					,(CAST(SUBSTR(ORIG_LN_ISS_DT, 1, 4) || '-' || SUBSTR(ORIG_LN_ISS_DT, 5, 2) || '-' || SUBSTR(ORIG_LN_ISS_DT, 7, 2) AS DATE)) AS EFF_DT
				FROM CORP.LOAN_DETAIL A
				JOIN PLAN_LIST B ON A.CASE_NO = B.ACCOUNT_NO
				
				) AS A
			WHERE EFF_DT BETWEEN CURRENT DATE - 14 MONTH AND CURRENT DATE
			GROUP BY CASE_NO
				,SOC_SEC_NO
				,LOAN_NO
				,EFF_DT
			) AS B ON A.CASE_NO = B.CASE_NO AND A.SOC_SEC_NO = B.SOC_SEC_NO AND A.LOAN_NO = B.LOAN_NO
		) AS B ON END_DATE >= B.EFF_DT
	)
	,WITH_PAYMENTS
AS (
	SELECT A.CASE_NO
		,A.SOC_SEC_NO
		,A.LOAN_NO
		,A.END_DATE
		,A.PRNC_BAL_DUE_AMT AS PRINC_BALANCE
	FROM LOAN_PAYMENTS A
	INNER JOIN (
		SELECT A.CASE_NO
			,A.SOC_SEC_NO
			,A.LOAN_NO
			,A.END_DATE
			,MIN(A.EFF_DT) AS EFF_DT
		FROM LOAN_PAYMENTS A
			,YEAR_BACK B
		WHERE A.EFF_DT BETWEEN B.BEGIN_DATE AND B.END_DATE AND A.END_DATE = B.END_DATE
		GROUP BY A.CASE_NO
			,A.SOC_SEC_NO
			,A.LOAN_NO
			,A.END_DATE
		) AS B ON A.LOAN_NO = B.LOAN_NO AND A.END_DATE = B.END_DATE AND A.EFF_DT = B.EFF_DT
	)
	,NO_PAYMENT
AS (
	SELECT A.CASE_NO
		,A.SOC_SEC_NO
		,A.LOAN_NO
		,A.END_DATE
		,A.PRNC_BAL_DUE_AMT AS PRINC_BALANCE
	FROM LOAN_PAYMENTS AS A
	INNER JOIN (
		SELECT A.BEGIN_DATE
			,A.END_DATE
			,A.LOAN_NO
			,MAX(A.EFF_DT) AS EFF_DT
		FROM LOAN_PAYMENTS A
		WHERE BEGIN_DATE >= EFF_DT
		GROUP BY A.BEGIN_DATE
			,A.END_DATE
			,A.LOAN_NO
		) AS B ON A.LOAN_NO = B.LOAN_NO AND A.END_DATE = B.END_DATE AND A.EFF_DT = B.EFF_DT
	WHERE NOT EXISTS (
			SELECT ''
			FROM WITH_PAYMENTS C
			WHERE A.LOAN_NO = C.LOAN_NO AND A.END_DATE = C.END_DATE
			)
	)
SELECT ' ' as new_case_no, ' ' as new_sub_id
	,substr(a.SOC_SEC_NO,1,3)||substr(a.SOC_SEC_NO,5,2)||substr(a.SOC_SEC_NO,8,4) as SSN
	,A.END_DATE AS EFF_DT
	,SUM(A.PRINC_BALANCE) HOLB, substr(a.case_NO,1,instr(a.case_no,' ')-1) as old_cont,SUBSTR(a.CASE_NO,11,5) old_CASE_NO
FROM (
	SELECT A.CASE_NO
		,A.SOC_SEC_NO
		,A.LOAN_NO
		,A.END_DATE
		,A.PRINC_BALANCE
	FROM NO_PAYMENT A
	
	UNION
	
	SELECT A.CASE_NO
		,A.SOC_SEC_NO
		,A.LOAN_NO
		,A.END_DATE
		,A.PRINC_BALANCE
	FROM WITH_PAYMENTS A
	) AS A
GROUP BY SUBSTR(A.CASE_NO,11,5), A.SOC_SEC_NO, A.END_DATE,substr(a.case_NO,1,instr(a.case_no,' ')-1)

FOR FETCH ONLY WITH UR
;
